{{- /* A file tree container. Overrides Hextra's filetree/container. The nested filetree/folder and filetree/file shortcodes only register themselves; this container draws the collected tree as a box-and-line diagram (partials/filetree/, styles in assets/css/custom.css). Folders are boxes with a tab, files boxes with a folded corner, joined by right-angled lines. The top-level folder's contents are spread side by side in source order; a run of consecutive files and collapsed folders shares one stacked column to save width. Every folder with contents opens and closes in place; state="closed" folders start closed. Usage is unchanged from Hextra's filetree. @example {{< filetree/container >}}{{< /filetree/container >}} */ -}} {{- $_ := .Inner -}} {{- $roots := .Store.Get "ft-children" | default slice -}}
{{- if eq (len $roots) 1 -}} {{- partial "filetree/node.html" (dict "node" (index $roots 0) "depth" 0 "root" true) -}} {{- else -}} {{- partial "filetree/group.html" (dict "children" $roots "depth" 0 "root" true) -}} {{- end -}}
